		function calculate() {
			const wordCount = document.getElementById("wordCount").value;
			const slowTime = (wordCount / 100);
			const normalTime = (wordCount / 150);
			const fastTime = (wordCount / 200);
			document.getElementById("slowTime").innerHTML = formatTime(slowTime);
			document.getElementById("normalTime").innerHTML = formatTime(normalTime);
			document.getElementById("fastTime").innerHTML = formatTime(fastTime);

		function formatTime(time) {
			const minutes = Math.floor(time);
			const seconds = Math.round((time - minutes) * 60);
			return minutes + " minute" + (minutes == 1 ? "" : "s") + ", " + seconds + " second" + (seconds == 1 ? "" : "s");
		}
